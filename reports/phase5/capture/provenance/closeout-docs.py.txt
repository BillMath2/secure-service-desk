from pathlib import Path
import json, hashlib, zipfile, shutil
root = Path.cwd()
def update(path, old, new):
    p = root / path
    text = p.read_text(encoding='utf-8')
    assert old in text, (path, old)
    p.write_text(text.replace(old, new), encoding='utf-8')

       'The final comparison corpus, repeated evaluation, final owner risk decisions, clean-checkout verification and recorded demo remain outstanding.',

p = root / 'reports/phase5/capture/results.json'
capture = json.loads(p.read_text())
capture['visual_review'] = 'Passed: all eight extracted segment frames inspected for legibility, missing text and clipping'
capture['audio_alignment'] = 'reports/phase5/capture/audio-alignment.json'
capture['passed'] = True
capture['inputs_sha256'] = {s: hashlib.sha256((root/s).read_bytes()).hexdigest() for s in (
    'docs/demo/narration-final.txt', 'docs/demo/storyboard.json', 'docs/demo/record-demo.mjs',
    '.lab/demo/narration-final.wav', 'reports/phase5/demo-evidence.json')}
p.write_text(json.dumps(capture, indent=2)+'\n')
shutil.copy2(root / '.lab/demo/check-audio-alignment.py',
             root / 'reports/phase5/capture/provenance/check-audio-alignment.py.txt')
bundle = root / '.lab/demo/secure-service-desk-demo-assets.zip'
assert not bundle.exists()
assets = [root / f'.lab/demo/slide-{i:02}.png' for i in range(1,9)]
assets += [root / f'.lab/demo/voice-{i:02}.wav' for i in range(1,9)]
assets += [root / '.lab/demo/narration-final.wav', root / 'docs/demo/storyboard.json',
           root / 'docs/demo/narration-final.txt', root / 'docs/demo/presentation-template.html']
with zipfile.ZipFile(bundle, 'x', compression=zipfile.ZIP_DEFLATED) as archive:
    for asset in assets:
        archive.write(asset, asset.relative_to(root).as_posix())
capture['asset_bundle'] = {'path': bundle.relative_to(root).as_posix(),
    'sha256': hashlib.sha256(bundle.read_bytes()).hexdigest(), 'bytes': bundle.stat().st_size}
p.write_text(json.dumps(capture, indent=2)+'\n')

p = root / 'reports/phase5/results.json'
result = json.loads(p.read_text())
result['technical_delivery_complete'] = True
result['delivery_evidence'] = {'release': 'reports/phase5/release/results.json',
    'capture': 'reports/phase5/capture/results.json', 'owner_review': 'docs/phase5-owner-review.md'}
for s in ('reports/phase5/release/results.json', 'reports/phase5/capture/results.json',
          'reports/phase5/clean-checkout-02/byte-preservation.json', 'reports/phase5/demo-evidence.json'):
    result['evidence_sha256'][s] = hashlib.sha256((root/s).read_bytes()).hexdigest()
assert result['initial_matched_gate_passed'] is False
assert result['owner_acceptance'] == 'pending' and result['phase5_exit_complete'] is False
p.write_text(json.dumps(result, indent=2)+'\n')

p = root / 'config/risk-register.json'
register = json.loads(p.read_text())
for risk in register['risks']:
    if int(risk['id'].split('-')[1]) <= 14:
        risk['evidence'].append('reports/phase5/release/results.json')
    if risk['id'] == 'RISK-14':
        risk['evidence'].append('reports/phase5/capture/results.json')
p.write_text(json.dumps(register, indent=2, ensure_ascii=False)+'\n', encoding='utf-8')
print('Final documentation, capture manifest and asset bundle written; owner decision remains pending.')
